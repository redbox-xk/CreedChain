# ============================================================================
# CRYPTOGRAPHIC PRIMITIVES (Bitcoin-compatible)
# ============================================================================

class BitcoinCrypto:
    """Bitcoin-compatible cryptographic functions"""
    
    @staticmethod
    def double_sha256(data: bytes) -> bytes:
        """Double SHA256 - Bitcoin's hash function"""
        return hashlib.sha256(hashlib.sha256(data).digest()).digest()
    
    @staticmethod
    def ripemd160_sha256(data: bytes) -> bytes:
        """RIPEMD160(SHA256(data)) for Bitcoin addresses"""
        sha256 = hashlib.sha256(data).digest()
        ripemd = hashlib.new('ripemd160')
        ripemd.update(sha256)
        return ripemd.digest()
    
    @staticmethod
    def base58_encode(data: bytes) -> str:
        """Base58 encoding (Bitcoin style)"""
        alphabet = '123456789ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz'
        n = int.from_bytes(data, 'big')
        encoded = ''
        while n > 0:
            n, rem = divmod(n, 58)
            encoded = alphabet[rem] + encoded
        
        # Add leading '1's for each leading zero byte
        leading_zeros = len(data) - len(data.lstrip(b'\x00'))
        return '1' * leading_zeros + encoded
    
    @staticmethod
    def base58_check_encode(data: bytes, prefix: int) -> str:
        """Base58Check encoding with version prefix"""
        version_data = bytes([prefix]) + data
        checksum = BitcoinCrypto.double_sha256(version_data)[:4]
        return BitcoinCrypto.base58_encode(version_data + checksum)
    
    @staticmethod
    def generate_keypair() -> Tuple[str, str]:
        """Generate Bitcoin-compatible keypair"""
        # Generate private key
        private_key = SigningKey.generate(curve=SECP256k1)
        private_key_hex = private_key.to_string().hex()
        
        # Get public key
        public_key = private_key.get_verifying_key()
        public_key_bytes = b'\x04' + public_key.to_string()  # Uncompressed
        
        # Create Bitcoin address
        pubkey_hash = BitcoinCrypto.ripemd160_sha256(public_key_bytes)
        address = BitcoinCrypto.base58_check_encode(pubkey_hash, BitcoinConstants.PUBKEY_HASH_PREFIX)
        
        # Create WIF (Wallet Import Format)
        wif = BitcoinCrypto.base58_check_encode(
            bytes([0x80]) + bytes.fromhex(private_key_hex) + bytes([0x01]),  # Compressed WIF
            0x80
        )
        
        return wif, address
    
    @staticmethod
    def sign_message(private_key_hex: str, message: bytes) -> str:
        """Sign message with Bitcoin-compatible signature"""
        private_key = SigningKey.from_string(bytes.fromhex(private_key_hex), curve=SECP256k1)
        signature = private_key.sign_deterministic(message, hashfunc=hashlib.sha256)
        return signature.hex()
    
    @staticmethod
    def verify_signature(public_key_hex: str, message: bytes, signature_hex: str) -> bool:
        """Verify Bitcoin signature"""
        try:
            public_key = VerifyingKey.from_string(bytes.fromhex(public_key_hex), curve=SECP256k1)
            signature = bytes.fromhex(signature_hex)
            return public_key.verify(signature, message, hashfunc=hashlib.sha256)
        except:
            return False

@dataclass
class TxInput:
    """Transaction Input (like Bitcoin)"""
    txid: str          # Previous transaction ID
    vout: int          # Output index in previous transaction
    script_sig: str    # Unlocking script
    sequence: int = 0xffffffff
    
    def serialize(self) -> bytes:
        """Serialize input for hashing"""
        txid_bytes = bytes.fromhex(self.txid)[::-1]  # Bitcoin uses little-endian for txid
        data = txid_bytes + struct.pack('<I', self.vout)
        script_sig_bytes = bytes.fromhex(self.script_sig)
        data += struct.pack('<B', len(script_sig_bytes)) + script_sig_bytes
        data += struct.pack('<I', self.sequence)
        return data
    
    def to_dict(self):
        return {
            'txid': self.txid,
            'vout': self.vout,
            'scriptSig': {
                'asm': self.script_sig,
                'hex': self.script_sig
            },
            'sequence': self.sequence
        }

@dataclass
class TxOutput:
    """Transaction Output (like Bitcoin)"""
    value: int           # Satoshis
    script_pubkey: str   # Locking script
    
    def serialize(self) -> bytes:
        """Serialize output for hashing"""
        data = struct.pack('<Q', self.value)
        script_bytes = bytes.fromhex(self.script_pubkey)
        data += struct.pack('<B', len(script_bytes)) + script_bytes
        return data
    
    def to_dict(self):
        return {
            'value': self.value / 100000000.0,  # Convert to BTC
            'n': 0,  # Will be set by transaction
            'scriptPubKey': {
                'asm': self.script_pubkey,
                'hex': self.script_pubkey
            }
        }

@dataclass
class Transaction:
    """Bitcoin-like Transaction"""
    version: int = 1
    inputs: List[TxInput] = None
    outputs: List[TxOutput] = None
    locktime: int = 0
    txid: str = None
    
    def __post_init__(self):
        if self.inputs is None:
            self.inputs = []
        if self.outputs is None:
            self.outputs = []
        if self.txid is None:
            self.txid = self.calculate_hash()
    
    def calculate_hash(self) -> str:
        """Calculate transaction ID (double SHA256)"""
        # Bitcoin uses different hashing for different purposes
        # For txid, it's double SHA256 of serialized transaction
        serialized = self.serialize_for_hash()
        return BitcoinCrypto.double_sha256(serialized).hex()
    
    def serialize_for_hash(self) -> bytes:
        """Serialize transaction for hashing (not including witnesses)"""
        data = struct.pack('<I', self.version)
        
        # Input count
        data += struct.pack('<B', len(self.inputs))
        
        # Inputs
        for txin in self.inputs:
            data += txin.serialize()
        
        # Output count
        data += struct.pack('<B', len(self.outputs))
        
        # Outputs
        for txout in self.outputs:
            data += txout.serialize()
        
        data += struct.pack('<I', self.locktime)
        return data
    
    def is_coinbase(self) -> bool:
        """Check if this is a coinbase transaction"""
        return len(self.inputs) == 1 and self.inputs[0].txid == '0' * 64
    
    def get_output_value(self, output_index: int) -> int:
        """Get value of specific output in satoshis"""
        if 0 <= output_index < len(self.outputs):
            return self.outputs[output_index].value
        return 0
    
    def to_dict(self):
        """Convert to Bitcoin-like JSON format"""
        return {
            'txid': self.txid,
            'hash': self.txid,  # Same as txid for non-segwit
            'version': self.version,
            'size': len(self.serialize_for_hash()),
            'locktime': self.locktime,
            'vin': [inp.to_dict() for inp in self.inputs],
            'vout': [out.to_dict() for i, out in enumerate(self.outputs)],
            'hex': self.serialize_for_hash().hex()
        }

@dataclass
class BlockHeader:
    """Bitcoin Block Header"""
    version: int
    prev_hash: str
    merkle_root: str
    timestamp: int
    bits: int  # Compact difficulty target
    nonce: int
    hash: str = None
    
    def __post_init__(self):
        if self.hash is None:
            self.hash = self.calculate_hash()
    
    def calculate_hash(self) -> str:
        """Calculate block hash (double SHA256 of header)"""
        # Serialize header (80 bytes like Bitcoin)
        data = struct.pack('<I', self.version)
        data += bytes.fromhex(self.prev_hash)[::-1]  # Little-endian
        data += bytes.fromhex(self.merkle_root)[::-1]
        data += struct.pack('<I', self.timestamp)
        data += struct.pack('<I', self.bits)
        data += struct.pack('<I', self.nonce)
        
        # Double SHA256
        hash_bytes = BitcoinCrypto.double_sha256(data)
        return hash_bytes[::-1].hex()  # Convert to big-endian for display
    
    def serialize(self) -> bytes:
        """Serialize block header"""
        data = struct.pack('<I', self.version)
        data += bytes.fromhex(self.prev_hash)[::-1]
        data += bytes.fromhex(self.merkle_root)[::-1]
        data += struct.pack('<I', self.timestamp)
        data += struct.pack('<I', self.bits)
        data += struct.pack('<I', self.nonce)
        return data
    
    def get_target(self) -> int:
        """Convert compact bits to target difficulty"""
        exponent = self.bits >> 24
        coefficient = self.bits & 0xffffff
        return coefficient * (256 ** (exponent - 3))
    
    def check_pow(self) -> bool:
        """Check if block satisfies Proof-of-Work"""
        target = self.get_target()
        hash_int = int(self.hash, 16)
        return hash_int <= target

@dataclass
class Block:
    """Bitcoin-like Block"""
    header: BlockHeader
    transactions: List[Transaction]
    height: int
    
    def calculate_merkle_root(self) -> str:
        """Calculate Merkle root of transactions"""
        if not self.transactions:
            return '0' * 64
        
        # Start with transaction hashes
        tx_hashes = [bytes.fromhex(tx.txid)[::-1] for tx in self.transactions]
        
        while len(tx_hashes) > 1:
            # If odd number, duplicate last
            if len(tx_hashes) % 2 != 0:
                tx_hashes.append(tx_hashes[-1])
            
            # Pair and hash
            new_hashes = []
            for i in range(0, len(tx_hashes), 2):
                pair_hash = BitcoinCrypto.double_sha256(tx_hashes[i] + tx_hashes[i + 1])
                new_hashes.append(pair_hash)
            
            tx_hashes = new_hashes
        
        return tx_hashes[0][::-1].hex()
    
    def to_dict(self):
        """Convert to Bitcoin-like JSON format"""
        return {
            'hash': self.header.hash,
            'height': self.height,
            'version': self.header.version,
            'previousblockhash': self.header.prev_hash,
            'merkleroot': self.header.merkle_root,
            'time': self.header.timestamp,
            'bits': hex(self.header.bits),
            'nonce': self.header.nonce,
            'size': 80 + sum(len(tx.serialize_for_hash()) for tx in self.transactions),
            'tx': [tx.txid for tx in self.transactions],
            'nTx': len(self.transactions),
            'difficulty': self.get_difficulty()
        }
    
    def get_difficulty(self) -> float:
        """Calculate block difficulty"""
        target = self.header.get_target()
        max_target = 0xffff * 256 ** (0x1d - 3)
        return max_target / target

class BitcoinBlockchain:
    """Bitcoin-like blockchain implementation"""
    
    def __init__(self, network: str = 'mainnet'):
        self.chain: List[Block] = []
        self.utxo_set: Dict[str, Dict[int, TxOutput]] = {}  # txid -> {vout: TxOutput}
        self.mempool: List[Transaction] = []
        self.difficulty = 0x1d00ffff  # Initial Bitcoin difficulty
        self.height = 0
        self.network = network
        
        # Coinbase maturity and subsidy
        self.coinbase_maturity = BitcoinConstants.COINBASE_MATURITY
        self.subsidy = BitcoinConstants.INITIAL_SUBSIDY
        self.halving_interval = BitcoinConstants.SUBSIDY_HALVING_INTERVAL
        
        # Initialize with genesis block
        self.create_genesis_block()
    
    def create_genesis_block(self):
        """Create Bitcoin-like genesis block"""
        # Genesis transaction (coinbase)
        coinbase_tx = Transaction(
            version=1,
            inputs=[
                TxInput(
                    txid='0' * 64,
                    vout=0xffffffff,
                    script_sig='04ffff001d0104455468652054696d65732030332f4a616e2f32303039204368616e63656c6c6f72206f6e206272696e6b206f66207365636f6e64206261696c6f757420666f722062616e6b73'
                )
            ],
            outputs=[
                TxOutput(
                    value=self.subsidy,
                    script_pubkey='4104678afdb0fe5548271967f1a67130b7105cd6a828e03909a67962e0ea1f61deb649f6bc3f4cef38c4f35504e51ec112de5c384df7ba0b8d578a4c702b6bf11d5fac'
                )
            ]
        )
        
        # Genesis block header
        genesis_header = BlockHeader(
            version=1,
            prev_hash='0' * 64,
            merkle_root=coinbase_tx.txid,
            timestamp=1231006505,  # Bitcoin genesis timestamp
            bits=0x1d00ffff,
            nonce=2083236893
        )
        
        # Create block
        genesis_block = Block(
            header=genesis_header,
            transactions=[coinbase_tx],
            height=0
        )
        
        # Add to chain
        self.chain.append(genesis_block)
        
        # Add UTXOs
        self.update_utxo_set(genesis_block)
        
        print(f"✅ Created Genesis Block (Bitcoin-like)")
        print(f"   Hash: {genesis_header.hash}")
        print(f"   Timestamp: {datetime.fromtimestamp(genesis_header.timestamp)}")
        print(f"   Difficulty: {genesis_block.get_difficulty():.2f}")
    
    def update_utxo_set(self, block: Block):
        """Update UTXO set with block's transactions"""
        for tx in block.transactions:
            # Remove spent UTXOs
            for txin in tx.inputs:
                if txin.txid != '0' * 64:  # Not coinbase
                    if txin.txid in self.utxo_set:
                        if txin.vout in self.utxo_set[txin.txid]:
                            del self.utxo_set[txin.txid][txin.vout]
            
            # Add new UTXOs
            if tx.txid not in self.utxo_set:
                self.utxo_set[tx.txid] = {}
            
            for i, txout in enumerate(tx.outputs):
                self.utxo_set[tx.txid][i] = txout
    
    def validate_transaction(self, tx: Transaction) -> bool:
        """Validate transaction (Bitcoin rules)"""
        # 1. Check basic structure
        if not tx.inputs or not tx.outputs:
            return False
        
        # 2. Check for double spend
        input_value = 0
        for txin in tx.inputs:
            if txin.txid in self.utxo_set and txin.vout in self.utxo_set[txin.txid]:
                input_value += self.utxo_set[txin.txid][txin.vout].value
            else:
                # Check mempool for unconfirmed parents
                found = False
                for mempool_tx in self.mempool:
                    if mempool_tx.txid == txin.txid:
                        found = True
                        break
                if not found:
                    return False
        
        # 3. Check output values
        output_value = sum(txout.value for txout in tx.outputs)
        
        # 4. Check fees (output <= input)
        if output_value > input_value and not tx.is_coinbase():
            return False
        
        # 5. Check dust outputs (minimum 546 satoshis)
        for txout in tx.outputs:
            if txout.value < 546:
                return False
        
        return True
    
    def validate_block(self, block: Block) -> bool:
        """Validate block (Bitcoin consensus rules)"""
        # 1. Check block hash meets difficulty
        if not block.header.check_pow():
            print("❌ Block doesn't meet Proof-of-Work")
            return False
        
        # 2. Check timestamp (not too far in future)
        current_time = int(time.time())
        if block.header.timestamp > current_time + 7200:  # 2 hours
            print("❌ Block timestamp too far in future")
            return False
        
        # 3. Check merkle root
        calculated_root = block.calculate_merkle_root()
        if calculated_root != block.header.merkle_root:
            print(f"❌ Invalid merkle root: {calculated_root} != {block.header.merkle_root}")
            return False
        
        # 4. Validate all transactions
        coinbase_count = 0
        for tx in block.transactions:
            if tx.is_coinbase():
                coinbase_count += 1
                if coinbase_count > 1:
                    print("❌ Multiple coinbase transactions")
                    return False
            elif not self.validate_transaction(tx):
                print(f"❌ Invalid transaction: {tx.txid}")
                return False
        
        # 5. Check block size
        block_size = 80 + sum(len(tx.serialize_for_hash()) for tx in block.transactions)
        if block_size > BitcoinConstants.MAX_BLOCK_SIZE:
            print(f"❌ Block too large: {block_size} bytes")
            return False
        
        return True
    
    def add_block(self, block: Block) -> bool:
        """Add validated block to chain"""
        if not self.validate_block(block):
            return False
        
        # Check previous block hash
        if block.header.prev_hash != self.chain[-1].header.hash:
            print("❌ Previous block hash doesn't match")
            return False
        
        # Update chain
        block.height = len(self.chain)
        self.chain.append(block)
        
        # Update UTXO set
        self.update_utxo_set(block)
        
        # Remove transactions from mempool
        self.mempool = [tx for tx in self.mempool if tx.txid not in [b.txid for b in block.transactions]]
        
        # Adjust difficulty every 2016 blocks
        if block.height % BitcoinConstants.DIFFICULTY_ADJUSTMENT_INTERVAL == 0:
            self.adjust_difficulty()
        
        # Halve subsidy every 210,000 blocks
        if block.height % self.halving_interval == 0:
            self.subsidy = self.subsidy // 2
        
        print(f"✅ Block #{block.height} added to chain")
        print(f"   Hash: {block.header.hash[:16]}...")
        print(f"   Transactions: {len(block.transactions)}")
        print(f"   Difficulty: {block.get_difficulty():.2f}")
        
        return True
    
    def adjust_difficulty(self):
        """Adjust difficulty based on actual vs expected time"""
        if len(self.chain) < BitcoinConstants.DIFFICULTY_ADJUSTMENT_INTERVAL:
            return
        
        # Get blocks from this period
        start_block = self.chain[-BitcoinConstants.DIFFICULTY_ADJUSTMENT_INTERVAL]
        end_block = self.chain[-1]
        
        # Calculate actual time taken
        actual_time = end_block.header.timestamp - start_block.header.timestamp
        expected_time = BitcoinConstants.TARGET_TIMESPAN
        
        # Adjust difficulty
        adjustment = expected_time / actual_time
        adjustment = max(0.25, min(4.0, adjustment))  # Limit adjustment
        
        new_target = int(self.difficulty * adjustment)
        self.difficulty = max(0x0000ffff, min(0x1d00ffff, new_target))
        
        print(f"📊 Difficulty adjusted: {adjustment:.2f}x")
        print(f"   New target: {hex(self.difficulty)}")
    
    def mine_block(self, miner_address: str) -> Optional[Block]:
        """Mine a new block (Proof-of-Work)"""
        if not self.mempool:
            print("⚠️  No transactions to mine")
            return None
        
        # Select transactions for block
        selected_txs = []
        total_size = 0
        for tx in self.mempool:
            tx_size = len(tx.serialize_for_hash())
            if total_size + tx_size + 80 <= BitcoinConstants.MAX_BLOCK_SIZE:
                selected_txs.append(tx)
                total_size += tx_size
        
        # Create coinbase transaction
        coinbase_tx = Transaction(
            inputs=[
                TxInput(
                    txid='0' * 64,
                    vout=0xffffffff,
                    script_sig=f'CREEDchain Block {len(self.chain)}'
                )
            ],
            outputs=[
                TxOutput(
                    value=self.subsidy,
                    script_pubkey=self.create_p2pkh_script(miner_address)
                )
            ]
        )
        
        # Add coinbase as first transaction
        transactions = [coinbase_tx] + selected_txs
        
        # Create block header
        header = BlockHeader(
            version=1,
            prev_hash=self.chain[-1].header.hash,
            merkle_root='',  # Will be calculated
            timestamp=int(time.time()),
            bits=self.difficulty,
            nonce=0
        )
        
        # Create block
        block = Block(
            header=header,
            transactions=transactions,
            height=len(self.chain)
        )
        
        # Calculate merkle root
        block.header.merkle_root = block.calculate_merkle_root()
        
        # Start mining (Proof-of-Work)
        print(f"⛏️  Mining block #{block.height}...")
        print(f"   Target: {hex(block.header.get_target())}")
        print(f"   Transactions: {len(transactions)}")
        
        start_time = time.time()
        while True:
            # Try nonce
            block.header.hash = block.header.calculate_hash()
            
            if block.header.check_pow():
                break
            
            # Increment nonce
            block.header.nonce += 1
            
            # If nonce exhausted, update timestamp
            if block.header.nonce >= 0xffffffff:
                block.header.timestamp = int(time.time())
                block.header.nonce = 0
        
        mining_time = time.time() - start_time
        print(f"✅ Block mined in {mining_time:.2f} seconds")
        print(f"   Hash: {block.header.hash}")
        print(f"   Nonce: {block.header.nonce}")
        
        return block
    
    def create_p2pkh_script(self, address: str) -> str:
        """Create P2PKH scriptPubKey"""
        # Decode address to get pubkey hash
        # For simplicity, we'll use a placeholder
        return f"76a914{address[-40:]}88ac"
    
    def get_balance(self, address: str) -> int:
        """Get balance for address (in satoshis)"""
        balance = 0
        
        for txid, outputs in self.utxo_set.items():
            for vout, txout in outputs.items():
                # Check if output belongs to address
                # In real implementation, you would decode scriptPubKey
                if address in txout.script_pubkey:
                    balance += txout.value
        
        return balance
    
    def get_blockchain_info(self) -> Dict:
        """Get blockchain information (Bitcoin-like)"""
        total_supply = sum(
            block.transactions[0].outputs[0].value  # Coinbase output
            for block in self.chain
            if block.transactions and block.transactions[0].is_coinbase()
        )
        
        return {
            'chain': self.network,
            'blocks': len(self.chain),
            'headers': len(self.chain),
            'bestblockhash': self.chain[-1].header.hash if self.chain else '0' * 64,
            'difficulty': self.chain[-1].get_difficulty() if self.chain else 1.0,
            'mediantime': self.chain[-1].header.timestamp if self.chain else int(time.time()),
            'verificationprogress': 1.0,
            'chainwork': hex(len(self.chain) * 0x1000),
            'pruned': False,
            'size_on_disk': sum(80 + sum(len(tx.serialize_for_hash()) for tx in block.transactions) for block in self.chain),
            'total_supply': total_supply / 100000000.0,  # In BTC
            'mempool_size': len(self.mempool),

class BitcoinP2PNode:
    """Bitcoin P2P network node"""
    
    def __init__(self, blockchain: BitcoinBlockchain, host: str = '0.0.0.0', port: int = 8333):
        self.blockchain = blockchain
        self.host = host
        self.port = port
        self.peers: Set[Tuple[str, int]] = set()
        self.running = False
        
        # Known DNS seeds (Bitcoin mainnet)
        self.dns_seeds = [
            'seed.bitcoin.sipa.be',
            'dnsseed.bluematt.me',
            'dnsseed.bitcoin.dashjr.org',
            'seed.bitcoinstats.com',
            'seed.bitcoin.jonasschnelli.ch'
        ]
    
    async def start(self):
        """Start P2P node"""
        self.running = True
        
        # Start server
        server = await asyncio.start_server(
            self.handle_connection,
            self.host,
            self.port
        )
        
        print(f"🌐 Bitcoin P2P Node started on {self.host}:{self.port}")
        
        # Connect to DNS seeds
        asyncio.create_task(self.discover_peers())
        
        async with server:
            await server.serve_forever()
    
    async def discover_peers(self):
        """Discover peers from DNS seeds"""
        import socket
        
        for seed in self.dns_seeds:
            try:
                addr_info = await asyncio.get_event_loop().getaddrinfo(
                    seed, self.port, proto=socket.IPPROTO_TCP
                )
                
                for info in addr_info:
                    ip = info[4][0]
                    self.peers.add((ip, self.port))
                    print(f"🔗 Discovered peer: {ip}:{self.port}")
                    
                    # Connect to peer
                    asyncio.create_task(self.connect_to_peer(ip, self.port))
            
            except Exception as e:
                print(f"⚠️  Failed to query DNS seed {seed}: {e}")
    
    async def connect_to_peer(self, host: str, port: int):
        """Connect to peer"""
        try:
            reader, writer = await asyncio.open_connection(host, port)
            
            # Send version message
            await self.send_version(writer)
            
            # Handle messages
            await self.handle_peer_messages(reader, writer)
            
        except Exception as e:
            print(f"⚠️  Failed to connect to {host}:{port}: {e}")
            self.peers.discard((host, port))
    
    async def handle_connection(self, reader, writer):
        """Handle incoming connection"""
        addr = writer.get_extra_info('peername')
        print(f"🔗 New connection from {addr}")
        
        # Add to peers
        self.peers.add(addr)
        
        try:
            # Send version
            await self.send_version(writer)
            
            # Handle messages
            await self.handle_peer_messages(reader, writer)
            
        except Exception as e:
            print(f"⚠️  Connection error with {addr}: {e}")
        finally:
            writer.close()
            await writer.wait_closed()
            self.peers.discard(addr)
    
    async def send_version(self, writer):
        """Send version message (Bitcoin protocol)"""
        version_msg = {
            'version': BitcoinConstants.PROTOCOL_VERSION,
            'services': BitcoinConstants.SERVICES,
            'timestamp': int(time.time()),
            'addr_recv': {'services': 0, 'ip': '0.0.0.0', 'port': 0},
            'addr_from': {'services': BitcoinConstants.SERVICES, 'ip': self.host, 'port': self.port},
            'nonce': random.getrandbits(64),
            'user_agent': '/CREEDchain:0.1.0/',
            'start_height': len(self.blockchain.chain),
            'relay': True
        }
        
        await self.send_message(writer, 'version', version_msg)
    
    async def send_message(self, writer, command: str, payload: Dict):
        """Send Bitcoin protocol message"""
        # Serialize payload
        payload_json = json.dumps(payload).encode()
        
        # Create message header
        magic = BitcoinConstants.NETWORK_MAGIC
        command_bytes = command.ljust(12, '\x00').encode()
        length = struct.pack('<I', len(payload_json))
        checksum = BitcoinCrypto.double_sha256(payload_json)[:4]
        
        # Send message
        writer.write(magic + command_bytes + length + checksum + payload_json)
        await writer.drain()
    
    async def handle_peer_messages(self, reader, writer):
        """Handle messages from peer"""
        while self.running:
            try:
                # Read message header (24 bytes)
                header = await reader.read(24)
                if len(header) < 24:
                    break
                
                # Parse header
                magic = header[:4]
                command = header[4:16].strip(b'\x00').decode()
                length = struct.unpack('<I', header[16:20])[0]
                checksum = header[20:24]
                
                # Read payload
                payload = await reader.read(length)
                
                if len(payload) < length:
                    break
                
                # Verify checksum
                if BitcoinCrypto.double_sha256(payload)[:4] != checksum:
                    print(f"⚠️  Invalid checksum from peer")
                    continue
                
                # Handle message
                await self.process_message(command, payload, writer)
                
            except Exception as e:
                print(f"⚠️  Error reading message: {e}")
                break
    
    async def process_message(self, command: str, payload: bytes, writer):
        """Process Bitcoin protocol message"""
        try:
            payload_dict = json.loads(payload.decode())
            
            if command == 'version':
                # Send verack
                await self.send_message(writer, 'verack', {})
                
                # Request blocks if we're behind
                if payload_dict.get('start_height', 0) > len(self.blockchain.chain):
                    await self.send_getblocks(writer)
            
            elif command == 'getblocks':
                # Send inv with block hashes
                block_hashes = [block.header.hash for block in self.blockchain.chain[-10:]]
                await self.send_message(writer, 'inv', {'inventory': block_hashes})
            
            elif command == 'getdata':
                # Send requested data
                inventory = payload_dict.get('inventory', [])
                for item in inventory:
                    if len(item) == 64:  # Block hash
                        for block in self.blockchain.chain:
                            if block.header.hash == item:
                                await self.send_message(writer, 'block', block.to_dict())
                                break
        
        except json.JSONDecodeError:
            print(f"⚠️  Invalid JSON payload for command {command}")

class BIP39:
    """BIP39 Mnemonic implementation"""
    
    @staticmethod
    def generate_mnemonic(strength: int = 128) -> str:
        """Generate BIP39 mnemonic phrase"""
        if strength % 32 != 0:
            raise ValueError("Strength must be multiple of 32")
        
        # Generate random entropy
        entropy = secrets.token_bytes(strength // 8)
        
        # Calculate checksum
        hash_bytes = hashlib.sha256(entropy).digest()
        checksum_length = strength // 32
        checksum = bin(int.from_bytes(hash_bytes, 'big'))[2:].zfill(256)[:checksum_length]
        
        # Combine entropy and checksum
        entropy_bits = bin(int.from_bytes(entropy, 'big'))[2:].zfill(strength)
        combined_bits = entropy_bits + checksum
        
        # Split into 11-bit chunks
        chunks = [combined_bits[i:i+11] for i in range(0, len(combined_bits), 11)]
        
        # Convert to words
        words = [BIP39_WORDLIST[int(chunk, 2)] for chunk in chunks]
        
        return ' '.join(words)
    
    @staticmethod
    def mnemonic_to_seed(mnemonic: str, passphrase: str = '') -> bytes:
        """Convert mnemonic to seed (BIP39)"""
        # Normalize
        mnemonic_normalized = unicodedata.normalize('NFKD', mnemonic)
        passphrase_normalized = unicodedata.normalize('NFKD', passphrase)
        
        # Create salt
        salt = 'mnemonic' + passphrase_normalized
        
        # PBKDF2
        seed = hashlib.pbkdf2_hmac(
            'sha512',
            mnemonic_normalized.encode('utf-8'),
            salt.encode('utf-8'),
            2048,
            64
        )
        
        return seed
    
    @staticmethod
    def validate_mnemonic(mnemonic: str) -> bool:
        """Validate BIP39 mnemonic"""
        words = mnemonic.split()
        if len(words) not in [12, 15, 18, 21, 24]:
            return False
        
        # Convert words back to bits
        word_indices = []
        for word in words:
            try:
                index = BIP39_WORDLIST.index(word)
                word_indices.append(index)
            except ValueError:
                return False
        
        # Convert indices to bits
        total_bits = len(words) * 11
        checksum_length = total_bits // 33
        entropy_length = total_bits - checksum_length
        
        # Reconstruct bits
        combined_bits = ''
        for index in word_indices:
            combined_bits += bin(index)[2:].zfill(11)
        
        # Split into entropy and checksum
        entropy_bits = combined_bits[:entropy_length]
        checksum_bits = combined_bits[entropy_length:]
        
        # Calculate checksum
        entropy_bytes = int(entropy_bits, 2).to_bytes(entropy_length // 8, 'big')
        hash_bytes = hashlib.sha256(entropy_bytes).digest()
        calculated_checksum = bin(int.from_bytes(hash_bytes, 'big'))[2:].zfill(256)[:checksum_length]
        
        return checksum_bits == calculated_checksum

class BIP32:
    """BIP32 Hierarchical Deterministic Wallets"""
    
    def __init__(self, seed: bytes):
        self.seed = seed
        self.master_key, self.master_chain_code = self._derive_master_keys()
    
    def _derive_master_keys(self) -> Tuple[bytes, bytes]:
        """Derive master keys from seed"""
        # HMAC-SHA512
        hmac_result = hmac.new(b'Bitcoin seed', self.seed, hashlib.sha512).digest()
        
        master_private_key = hmac_result[:32]
        master_chain_code = hmac_result[32:]
        
        # Ensure private key is valid
        n = 0xFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFEBAAEDCE6AF48A03BBFD25E8CD0364141
        if int.from_bytes(master_private_key, 'big') >= n or int.from_bytes(master_private_key, 'big') == 0:
            raise ValueError("Invalid master private key")
        
        return master_private_key, master_chain_code
    
    def derive_path(self, path: str) -> Tuple[bytes, bytes]:
        """Derive key from path (e.g., "m/44'/0'/0'/0/0")"""
        if not path.startswith('m'):
            raise ValueError("Path must start with 'm'")
        
        # Parse path
        parts = path.split('/')[1:]  # Remove 'm'
        key = self.master_key
        chain_code = self.master_chain_code
        
        for part in parts:
            hardened = part.endswith("'")
            index = int(part.rstrip("'"))
            
            if hardened:
                index += 0x80000000
            
            key, chain_code = self._ckd_priv(key, chain_code, index)
        
        return key, chain_code
    
    def _ckd_priv(self, parent_key: bytes, parent_chain_code: bytes, index: int) -> Tuple[bytes, bytes]:
        """Child key derivation (private -> private)"""
        if index >= 0x80000000:  # Hardened
            data = b'\x00' + parent_key + struct.pack('>I', index)
        else:  # Normal
            # Get public key from private key
            private_key = ecdsa.SigningKey.from_string(parent_key, curve=SECP256k1)
            public_key = private_key.get_verifying_key()
            public_key_bytes = b'\x04' + public_key.to_string()
            
            data = public_key_bytes + struct.pack('>I', index)
        
        # HMAC-SHA512
        hmac_result = hmac.new(parent_chain_code, data, hashlib.sha512).digest()
        
        child_key = (int.from_bytes(hmac_result[:32], 'big') + int.from_bytes(parent_key, 'big')) % int(
            0xFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFEBAAEDCE6AF48A03BBFD25E8CD0364141
        )
        
        child_key_bytes = child_key.to_bytes(32, 'big')
        child_chain_code = hmac_result[32:]
        
        return child_key_bytes, child_chain_code
    
    def private_key_to_public_key(self, private_key: bytes) -> bytes:
        """Convert private key to uncompressed public key"""
        sk = ecdsa.SigningKey.from_string(private_key, curve=SECP256k1)
        vk = sk.get_verifying_key()
        return b'\x04' + vk.to_string()
    
    def private_key_to_wif(self, private_key: bytes, compressed: bool = True) -> str:
        """Convert private key to WIF format"""
        # Add version byte (0x80 for mainnet)
        extended_key = b'\x80' + private_key
        
        if compressed:
            extended_key += b'\x01'
        
        # Double SHA256 checksum
        checksum = hashlib.sha256(hashlib.sha256(extended_key).digest()).digest()[:4]
        
        # Base58Check encode
        return base58.b58encode(extended_key + checksum).decode()
    
    def public_key_to_address(self, public_key: bytes, compressed: bool = True) -> str:
        """Convert public key to Bitcoin address"""
        if compressed:
            # Compressed public key
            x = public_key[1:33]
            y = int.from_bytes(public_key[33:], 'big')
            prefix = b'\x02' if y % 2 == 0 else b'\x03'
            public_key = prefix + x
        
        # SHA256 then RIPEMD160
        sha256_hash = hashlib.sha256(public_key).digest()
        ripemd160_hash = hashlib.new('ripemd160', sha256_hash).digest()
        
        # Mainnet P2PKH address
        return base58.b58encode_check(b'\x00' + ripemd160_hash).decode()

class HDWallet:
    """Bitcoin HD Wallet Manager"""
    
    def __init__(self, mnemonic: str = None, passphrase: str = ''):
        if mnemonic is None:
            mnemonic = BIP39.generate_mnemonic(128)
        
        if not BIP39.validate_mnemonic(mnemonic):
            raise ValueError("Invalid mnemonic phrase")
        
        self.mnemonic = mnemonic
        self.passphrase = passphrase
        self.seed = BIP39.mnemonic_to_seed(mnemonic, passphrase)
        self.bip32 = BIP32(self.seed)
        
        # BIP44 paths
        self.paths = {
            'external': "m/44'/0'/0'/0",  # Receiving addresses
            'change': "m/44'/0'/0'/1",    # Change addresses
        }
        
        self.addresses = {}
        self.private_keys = {}
    
    def generate_address(self, account: int = 0, change: int = 0, index: int = 0) -> Dict:
        """Generate Bitcoin address at specified derivation path"""
        path = f"m/44'/0'/{account}'/{change}/{index}"
        
        # Derive keys
        private_key, chain_code = self.bip32.derive_path(path)
        public_key = self.bip32.private_key_to_public_key(private_key)
        
        # Convert to formats
        wif = self.bip32.private_key_to_wif(private_key)
        address = self.bip32.public_key_to_address(public_key)
        
        # Store
        self.addresses[address] = {
            'path': path,
            'public_key': public_key.hex(),
            'private_key': private_key.hex(),
            'wif': wif
        }
        
        self.private_keys[private_key.hex()] = address
        
        return {
            'address': address,
            'path': path,
            'public_key': public_key.hex(),
            'private_key': private_key.hex()[:16] + '...',  # Don't expose full private key
            'wif': wif[:16] + '...'
        }
    
    def generate_multiple_addresses(self, count: int = 10, change: int = 0) -> List[Dict]:
        """Generate multiple addresses"""
        addresses = []
        for i in range(count):
            addr = self.generate_address(change=change, index=i)
            addresses.append(addr)
        
        return addresses
    
    def get_address_info(self, address: str) -> Optional[Dict]:
        """Get information for address"""
        return self.addresses.get(address)
    
    def sign_transaction(self, private_key_hex: str, transaction_data: bytes) -> str:
        """Sign transaction with private key"""
        if private_key_hex not in self.private_keys:
            raise ValueError("Private key not found in wallet")
        
        private_key = ecdsa.SigningKey.from_string(
            bytes.fromhex(private_key_hex),
            curve=SECP256k1
        )
        
        signature = private_key.sign_deterministic(
            transaction_data,
            hashfunc=hashlib.sha256,
            sigencode=ecdsa.util.sigencode_der_canonize
        )
        
        return signature.hex()
    
    def export_wallet(self, password: str = None) -> Dict:
        """Export wallet data (encrypted if password provided)"""
        data = {
            'mnemonic': self.mnemonic,
            'passphrase': self.passphrase,
            'addresses': self.addresses,
            'created': int(time.time())
        }
        
        if password:
            # Encrypt with password
            cipher = BitcoinCrypto.encrypt_data(json.dumps(data), password)
            return {'encrypted': True, 'data': cipher}
        else:
            return {'encrypted': False, 'data': data}
    
    @staticmethod
    def import_wallet(data: Dict, password: str = None) -> 'HDWallet':
        """Import wallet from export data"""
        if data.get('encrypted'):
            if not password:
                raise ValueError("Password required for encrypted wallet")
            
            decrypted = BitcoinCrypto.decrypt_data(data['data'], password)
            wallet_data = json.loads(decrypted)
        else:
            wallet_data = data['data']
        
        return HDWallet(
            mnemonic=wallet_data['mnemonic'],
            passphrase=wallet_data.get('passphrase', '')
        )

class CPUMiner:
    """CPU-based Bitcoin miner"""
    
    def __init__(self, blockchain: BitcoinBlockchain, miner_address: str):
        self.blockchain = blockchain
        self.miner_address = miner_address
        self.mining = False
        self.hash_rate = 0
        self.shares_found = 0
        self.threads = []
        self.max_threads = 4  # CPU threads
    
    def mine_block(self) -> Optional[Block]:
        """Mine a single block"""
        start_time = time.time()
        hashes_tried = 0
        
        # Get block template
        block = self.blockchain.mine_block(self.miner_address)
        if not block:
            return None
        
        print(f"⛏️  Mining block #{block.height}")
        print(f"   Target: {hex(block.header.get_target())}")
        print(f"   Difficulty: {block.get_difficulty():.2f}")
        
        # Start mining loop
        while self.mining:
            # Try nonce
            block.header.hash = block.header.calculate_hash()
            hashes_tried += 1
            
            if block.header.check_pow():
                mining_time = time.time() - start_time
                self.hash_rate = hashes_tried / mining_time
                
                print(f"✅ Block found!")
                print(f"   Hash: {block.header.hash}")
                print(f"   Nonce: {block.header.nonce}")
                print(f"   Time: {mining_time:.2f}s")
                print(f"   Hash rate: {self.hash_rate:,.0f} H/s")
                print(f"   Shares: {self.shares_found}")
                
                return block
            
            # Increment nonce
            block.header.nonce += 1
            
            # If nonce exhausted, update timestamp
            if block.header.nonce >= 0xffffffff:
                block.header.timestamp = int(time.time())
                block.header.nonce = 0
        
        return None
    
    def start_mining(self):
        """Start mining in background"""
        if self.mining:
            print("⚠️  Miner already running")
            return
        
        self.mining = True
        print(f"⛏️  Starting miner for {self.miner_address}")
        print(f"   Threads: {self.max_threads}")
        
        # Start mining threads
        for i in range(self.max_threads):
            thread = threading.Thread(target=self._mining_thread, args=(i,))
            thread.daemon = True
            thread.start()
            self.threads.append(thread)
        
        print("✅ Miner started")
    
    def stop_mining(self):
        """Stop mining"""
        self.mining = False
        for thread in self.threads:
            thread.join(timeout=1)
        
        self.threads.clear()
        print("🛑 Miner stopped")
    
    def _mining_thread(self, thread_id: int):
        """Mining thread worker"""
        print(f"   Thread {thread_id} started")
        
        while self.mining:
            block = self.mine_block()
            if block:
                # Add block to blockchain
                success = self.blockchain.add_block(block)
                if success:
                    self.shares_found += 1
                    print(f"💰 Block #{block.height} accepted!")
                else:
                    print(f"❌ Block #{block.height} rejected")
                
                # Wait a bit before mining next block
                time.sleep(1)

        self.methods = {
            'getblockchaininfo': self.getblockchaininfo,
            'getnetworkinfo': self.getnetworkinfo,
            'getwalletinfo': self.getwalletinfo,
            'getblock': self.getblock,

            'getmempoolinfo': self.getmempoolinfo,

                'error': None,
                'id': 1

    def getblock(self, params: list) -> Dict[str, Any]:
        """getblock RPC method"""
        if not params:
            raise ValueError("Block hash or height required")
        
        identifier = params[0]
        
        # Try as height
        try:
            height = int(identifier)
            if 0 <= height < len(blockchain.chain):
                block = blockchain.chain[height]
                return block.to_dict()
        except ValueError:
            pass
        
        # Try as hash
        for block in blockchain.chain:
            if block.header.hash == identifier:
                return block.to_dict()
        
        raise ValueError(f"Block not found: {identifier}")
    
    def getblockhash(self, params: list) -> str:
        """getblockhash RPC method"""
        if not params:
            raise ValueError("Height required")
        
        height = int(params[0])
        if 0 <= height < len(blockchain.chain):
            return blockchain.chain[height].header.hash
        
        raise ValueError(f"Block not found at height {height}")
    
    def getrawtransaction(self, params: list) -> Dict[str, Any]:
        """getrawtransaction RPC method"""
        if not params:
            raise ValueError("Transaction ID required")
        
        txid = params[0]
        verbose = params[1] if len(params) > 1 else True
        
        # Search in blockchain
        for block in blockchain.chain:
            for tx in block.transactions:
                if tx.txid == txid:
                    if verbose:
                        return tx.to_dict()
                    else:
                        return tx.serialize_for_hash().hex()
        
        # Search in mempool
        for tx in blockchain.mempool:
            if tx.txid == txid:
                if verbose:
                    return tx.to_dict()
                else:
                    return tx.serialize_for_hash().hex()
        
        raise ValueError(f"Transaction not found: {txid}")

        """listtransactions RPC method"""
        account = params[0] if len(params) > 0 else "*"
        count = int(params[1]) if len(params) > 1 else 10
        skip = int(params[2]) if len(params) > 2 else 0
        
        # Collect transactions for all wallet addresses
        all_txs = []
        for address in wallet.addresses.keys():
            txs = blockchain.get_user_transactions(address)
            for tx in txs:
                tx['address'] = address
                all_txs.append(tx)
        
        # Sort by timestamp (newest first)
        all_txs.sort(key=lambda x: x.get('timestamp', 0), reverse=True)
        
        return all_txs[skip:skip + count]
